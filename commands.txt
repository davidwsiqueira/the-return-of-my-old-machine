# ============================================
# MkDocs Commands for "The Return of My Old Machine"
# ============================================

# Initial Setup (First Time Only)
# ============================================

# 1. Create virtual environment
python3 -m venv venv

# 2. Activate virtual environment
source venv/bin/activate

# 3. Install dependencies
pip install -r requirements.txt

# 4. Verify installation
mkdocs --version


# Daily Development Workflow
# ============================================

# Activate virtual environment (always do this first)
source venv/bin/activate

# Start development server with live reload
mkdocs serve

# Start server on specific port
mkdocs serve -a localhost:8080

# Start server accessible from network
mkdocs serve -a 0.0.0.0:8000


# Building and Deployment
# ============================================

# Build static site (creates 'site' directory)
mkdocs build

# Build with clean (removes old site directory first)
mkdocs build --clean

# Build with strict mode (warnings become errors)
mkdocs build --strict


# Content Management
# ============================================

# Create new MkDocs project (if starting fresh)
mkdocs new my-project

# Validate configuration
mkdocs build --strict


# GitHub Pages Deployment
# ============================================

# Deploy to GitHub Pages (builds and pushes to gh-pages branch)
mkdocs gh-deploy

# Deploy with clean build
mkdocs gh-deploy --clean

# Deploy with custom commit message
mkdocs gh-deploy -m "Updated documentation"


# Virtual Environment Management
# ============================================

# Deactivate virtual environment
deactivate

# Update pip
pip install --upgrade pip

# Update all packages
pip install --upgrade -r requirements.txt

# Freeze current dependencies
pip freeze > requirements.txt

# Remove virtual environment (if needed)
rm -rf venv


# Troubleshooting
# ============================================

# Clear MkDocs cache
rm -rf site/

# Reinstall dependencies
pip install --force-reinstall -r requirements.txt

# Check for broken links
mkdocs build --strict

# View detailed build output
mkdocs build --verbose

# Fix "Address already in use" error (port 8000 is busy)
# Option 1: Find and kill the process using port 8000
lsof -ti:8000 | xargs kill -9

# Option 2: Use a different port
mkdocs serve -a localhost:8080

# Option 3: Find what's using the port
lsof -i :8000

# Create the missing file or remove the links from your markdown files


# Quick Reference
# ============================================

# One-liner to activate and serve
source venv/bin/activate && mkdocs serve

# One-liner to activate, build, and deploy
source venv/bin/activate && mkdocs gh-deploy --clean

# Check if venv is active (should show path to venv)
which python

# - Changes to docs/ are auto-reloaded in dev server
